## 1. 项目概述

### 1.1 什么是知识卡片系统？

知识卡片系统是 IELTS Learning Agent 项目的一个功能模块，用于：
- 存储和管理雅思学习知识卡片（课程内容）
- 提供命令行界面（CLI）让用户浏览和下载课程
- 支持 Word 文档（.docx）转换为系统可读的 JSON 格式

def extract_topic_from_filename(filename: str) -> str:
    """
    从文件名提取主题名称
    
    输入: "Sentence01_太空探索_知识卡片.docx"
    输出: "太空探索"
    """
    # 正则表达式：匹配 "Sentence" + 数字 + "_" + 主题 + "_知识卡片"
    match = re.search(r' Sentence?\d+_(.+?)_知识卡片', filename)
    if match:
        return match.group(1).strip()
    return Path(filename).stem


# ============================================
# 2. 辅助函数：解析音标
# ============================================
def parse_phonetic(text: str) -> Optional[str]:
    """
    从文本中提取音标
    
    输入: "considerable /kənˈsɪdərəbl/"
    输出: "/kənˈsɪdərəbl/"
    """
    # 正则：匹配 /.../ 格式的内容
    match = re.search(r'/([^/]+)/', text)
    if match:
        return match.group(0)  # 保留完整格式
    return None

class CardParser:
    """知识卡片解析器 - 负责从 Word 文档中提取内容"""
    
    # 卡片标题的匹配模式（正则表达式 → 内部名称）
    CARD_PATTERNS = {
        r'卡片\s*(\d+)': 'card_header',      # 匹配 "卡片 1"
        r'核心例句': 'core_sentence',          # 匹配 "核心例句"
        r'语法笔记': 'grammar',                 # 匹配 "语法笔记"
        r'核心词表': 'vocabulary',             # 匹配 "核心词表"
        # ... 更多模式
    }
    
    def __init__(self, doc):
        """初始化解析器，传入 docx 文档对象"""
        self.doc = doc
        self.current_section = None  # 当前解析的章节
        self.content = {
            "topic": "",             # 主题
            "core_sentence": {},     # 核心例句
            "vocabulary": [],        # 词汇列表
            # ... 更多字段
        }
        self._parse()  # 开始解析
    
    def _parse(self):
        """解析文档的主方法"""
        # 遍历所有段落
        for para in self.doc.paragraphs:
            self._process_paragraph(para)
        
        # 遍历所有表格
        for table in self.doc.tables:
            table_data = self._parse_table(table)
            self._process_table(table_data)
    
    def _parse_table(self, table):
        """解析表格数据"""
        data = []
        for row in table.rows:           # 遍历每一行
            row_data = [cell.text.strip()  # 提取每个单元格文本
                        for cell in row.cells]
            data.append(row_data)
        return data  # 返回二维数组
    
    def _process_table(self, table_data):
        """处理表格数据（如词汇表）"""
        # ... 处理逻辑
        pass
    
    def _process_paragraph(self, para):
        """处理段落文本"""
        text = para.text.strip()
        if not text:
            return
        
        # 检测当前段落的类型
        for pattern, section in self.CARD_PATTERNS.items():
            if re.search(pattern, text, re.IGNORECASE):

def convert_docx_to_json(docx_path: Path, output_dir: Optional[Path] = None):
    """
    转换 docx 文件为主函数
    
    参数:
        docx_path: Word 文件路径
        output_dir: 输出目录（默认与 docx 同目录）
    
    返回:
        content 字典（课程内容）
    """
    # 1. 读取 Word 文档
    doc = Document(docx_path)
    
    # 2. 解析内容
    parser = CardParser(doc)
    parser.content["topic"] = extract_topic_from_filename(docx_path.name)
    
    # 3. 生成 JSON 文件
    content_json_path = output_dir / "content.json"

        "id": f"course_{topic}",
        "topic": topic,
        # ...
    }

class CourseService:
    """知识卡片课程服务"""
    
    def __init__(self, courses_dir: Path | None = None):
        """
        初始化服务
        
        参数:
            courses_dir: 课程目录路径，默认为项目根目录下的 courses/
        """
        if courses_dir is None:
            # 计算 courses 目录的路径
            # 从当前文件位置向上 3 级到项目根目录
            self._courses_dir = Path(__file__).resolve().parents[3] / "courses"
        else:
            self._courses_dir = courses_dir
    
    def list_courses(self) -> list[dict]:
        """
        列出所有课程
        
        返回:
            课程列表，每个课程是一个字典：
            [
                {"topic": "太空探索", "title": "...", "source": "..."},
                ...
            ]
        """
        courses = []
        
        # 遍历 courses 目录下的每个子目录
        for topic_dir in self._courses_dir.iterdir():
            if not topic_dir.is_dir():
                continue
            
            meta_path = topic_dir / "meta.json"
            if not meta_path.exists():
                continue
            
            # 读取并解析 JSON
            meta = json.loads(meta_path.read_text(encoding="utf-8"))
            
            courses.append({
                "topic": meta.get("topic", topic_dir.name),
                "title": meta.get("title", ""),
                "source": meta.get("source", ""),
                "path": str(topic_dir)
            })
        
        return sorted(courses, key=lambda x: x["topic"])
    
    def get_course(self, topic: str) -> Optional[dict]:
        """
        获取指定课程
        
        参数:
            topic: 课程主题名
        
        返回:
            课程数据字典，或 None（如果不存在）
        """
        # 精确匹配
        course_path = self._courses_dir / topic
        
        # 模糊匹配（如果精确匹配失败）

                return None

            meta = json.loads(meta_path.read_text(encoding="utf-8"))

            "topic": topic,
            "meta": meta,

@course.command("list")
def list_courses():
    """列出所有课程"""

    svc = CourseService()

    courses = svc.list_courses()
    
    if not courses:

        return

        title = c["title"][:28] if c["title"] else ""
        source = c["source"][:13] if c["source"] else ""
        click.echo(f"{topic:<20} {title:<30} {source:<15}")

# 3. show 命令：显示课程内容
# ============================================
@course.command("show")
@click.argument("topic")           # 位置参数：课程主题

    """显示课程内容"""
    svc = CourseService()
    
    # 尝试获取 Markdown 格式
    md = svc.get_markdown(topic)
    if md:
        import sys
        sys.stdout.write(md)      # 直接输出
        return
    
    # 回退到 JSON 格式
    course_data = svc.get_course(topic)
    if not course_data:
        click.echo(f"错误: 课程 '{topic}' 不存在", err=True)
        return
    
    # 格式化输出 JSON 数据
    content = course_data["content"]
    meta = course_data["meta"]
    
    click.echo(f"\n# {content.get('topic', topic)}\n")
    
    if meta.get("source"):
        click.echo(f"来源: {meta['source']}\n")
    
    # ... 更多输出逻辑


# ============================================
# 4. download 命令：下载原始文件
# ============================================
@course.command("download")
@click.argument("topic")
@click.option("--path", "-p", default=".", help="下载到指定目录")
def download_course(topic: str, path: str):
    """下载课程原始文件"""
    svc = CourseService()
    docx_path = svc.get_docx_path(topic)
    
    if not docx_path:
        click.echo(f"错误: 课程 '{topic}' 的原始文件不存在", err=True)
        return
    
    import shutil
    dest = Path(path) / docx_path.name
    shutil.copy2(docx_path, dest)
    click.echo(f"✓ 已下载到: {dest}")

@course.command("download-all")

def download_all(path: str):
