### 步骤 A：批量改名 20 个文件

把项目内所有 "雅思搭子" 替换为 "屠鸭口语磨炼"：
```bash
cd /Users/wangliping/Public/selfProgram/ielts_apply_seek
python3 -c "
import os
old = '雅思搭子'
new = '屠鸭口语磨炼'
count = 0
files = 0
for root, dirs, fs in os.walk('.'):
    if '.git' in root or '__pycache__' in root:
        continue
    for f in fs:
        if f.endswith(('.json', '.md', '.py', '.js', '.wxml', '.wxss', '.txt', '.sh', '.yml', '.yaml')):
            fp = os.path.join(root, f)
            try:
                content = open(fp, encoding='utf-8').read()
                if old in content:
                    new_content = content.replace(old, new)
                    open(fp, 'w', encoding='utf-8').write(new_content)
                    count += content.count(old)
                    files += 1
                    print(f'  ✅ {fp} ({content.count(old)} 处)')
            except: pass
print(f'\n共 {files} 个文件，{count} 处替换')
"
```
